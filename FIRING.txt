Обзор от ИИ              

To create a functioning 3D firing turret in Godot 4, structure your node hierarchy into a base (for horizontal rotation), a top or barrel (for vertical elevation), and a muzzle/spawn point. 
The script will track targets, rotate the nodes using look_at(), and instance projectiles.

1. Node Hierarchy
Set up your scene tree using Node3D nodes:

	Turret (Node3D): Holds the main script.
		Base (Node3D): Rotates strictly along the \(Y\)-axis.
			Top/Barrel (Node3D): Rotates strictly along the \(X\)-axis.
				Muzzle (Marker3D): The exact point where projectiles spawn.

2. Turret Tracking and Firing ScriptAttach this script to your root Turret node.

=================================================
extends Node3D

@export var projectile: PackedScene
@export var rotation_speed: float = 5.0
@export var fire_rate: float = 1.0

@onready var base: Node3D = $Base
@onready var top: Node3D = $Top
@onready var muzzle: Marker3D = $Base/Top/Muzzle
@onready var timer: Timer = $Timer

var current_target: Node3D = null

func _ready() -> void:
	timer.wait_time = fire_rate
	timer.timeout.connect(_on_fire_timer_timeout)
	timer.start()

func _physics_process(delta: float) -> void:
	current_target = get_closest_target()
	
	if current_target:
		# Calculate direction to target
		var target_pos = current_target.global_position
		
		# Rotate base (Yaw)
		var base_target_pos = Vector3(target_pos.x, base.global_position.y, target_pos.z)
		base.look_at(base_target_pos, Vector3.UP)
		
		# Rotate top/barrel (Pitch)
		var top_target_pos = Vector3(base.global_position.x, target_pos.y, base.global_position.z)
		top.look_at(top_target_pos, Vector3.UP)

func get_closest_target() -> Node3D:
	# Replace this with your own logic to scan/find enemies in an area
	var potential_targets = get_tree().get_nodes_in_group("enemies")
	var closest: Node3D = null
	var min_distance: float = 9999.0
	
	for target in potential_targets:
		var distance = global_position.distance_to(target.global_position)
		if distance < min_distance:
			min_distance = distance
			closest = target
			
	return closest

func _on_fire_timer_timeout() -> void:
	if current_target and projectile:
		var b = projectile.instantiate()
		get_tree().root.add_child(b)
		b.global_transform = muzzle.global_transform
=================================================

3. Basic Projectile Script
Create a separate scene for your bullet or projectile. Attach a script to make it travel forward:

=================================================
extends Node3D

@export var speed: float = 20.0

func _physics_process(delta: float) -> void:
	# Move forward in the direction the muzzle is pointing
	position -= transform.basis.z * speed * delta

func _on_body_entered(body: Node) -> void:
	# Deal damage to enemy
	if body.is_in_group("enemies"):
		body.queue_free() # Or call a take_damage() function
	queue_free()
